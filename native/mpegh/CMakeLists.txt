cmake_minimum_required(VERSION 3.15)
cmake_policy(SET CMP0091 NEW)
project(MpeghExport C)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(LC_LEVEL_4 ON CACHE BOOL "Level 4 decoder" FORCE)
set(MPEGH_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../.tools/libmpegh" CACHE PATH "Pinned clean source")
add_subdirectory("${MPEGH_SOURCE}" upstream EXCLUDE_FROM_ALL)
file(GLOB MP4_SOURCES "${MPEGH_SOURCE}/test/mp4/*.c")
add_executable(MpeghExport main.c ${MP4_SOURCES})
target_link_libraries(MpeghExport PRIVATE ia_mpeghd_lib)
target_include_directories(MpeghExport PRIVATE "${MPEGH_SOURCE}/test/mp4")
target_compile_definitions(MpeghExport PRIVATE LC_LEVEL_4 _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(MpeghExport PRIVATE /utf-8 /O2)
    # Level 4 initialisation/rendering uses large automatic work arrays.
    target_link_options(MpeghExport PRIVATE /STACK:16777216)
    target_compile_options(ia_mpeghd_lib PRIVATE /utf-8)
else()
    target_link_libraries(MpeghExport PRIVATE m)
endif()
