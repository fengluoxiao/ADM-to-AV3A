cmake_minimum_required(VERSION 3.24)
project(av3s_encoder C)
set(CMAKE_C_STANDARD 11)
set(AV3A_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.tools/av3a/av3adecoder" CACHE PATH "AVS3 RM source")
file(GLOB COMMON "${AV3A_SOURCE}/libavs3_common/*.c" "${AV3A_SOURCE}/libavs3_debug/*.c")
file(GLOB ENCODER "${AV3A_SOURCE}/avs3Encoder/src/*.c")
# Encoder-only build: decoder entropy synthesis is not part of this tool.
list(FILTER COMMON EXCLUDE REGEX "/mdct_dequant_decode\\.c$")
add_executable(avs3Encoder ${COMMON} ${ENCODER})
target_include_directories(avs3Encoder PRIVATE "${AV3A_SOURCE}/libavs3_common"
    "${AV3A_SOURCE}/libavs3_debug" "${AV3A_SOURCE}/avs3Encoder/include" "${AV3A_SOURCE}/avs3Decoder/include")
if(MSVC)
    set_property(TARGET avs3Encoder PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_compile_definitions(avs3Encoder PRIVATE WIN32 _CRT_SECURE_NO_WARNINGS)
    target_compile_options(avs3Encoder PRIVATE /O2 /utf-8 /Gy /FI${CMAKE_CURRENT_LIST_DIR}/msvc_compat.h)
    target_link_options(avs3Encoder PRIVATE /OPT:REF)
else()
    target_compile_options(avs3Encoder PRIVATE -O3)
    target_link_libraries(avs3Encoder PRIVATE m)
endif()
